<script>
const ADMIN_PASSWORD = "change-me";

const starterProducts = [
  {
    name: "Coming soon",
    price: "",
    description: "A new glow essential is on the way."
  }
];

let items =
  JSON.parse(localStorage.getItem("glow-products") || "null") ||
  starterProducts;

const escapeText = value =>
  String(value).replace(/[&<>"]/g, character => ({
    "&": "&amp;",
    "<": "&lt;",
    ">": "&gt;",
    '"': "&quot;"
  })[character]);

function renderProducts() {
  products.innerHTML = items.map(product => `
    <article class="product">
      <div class="product-img">✦</div>

      <div class="product-row">
        <div>
          <h3>${escapeText(product.name)}</h3>
          <p>${escapeText(product.description || "")}</p>
        </div>

        <strong>${escapeText(product.price || "")}</strong>
      </div>

      <a
        href="https://wa.me/96170887031?text=${
          encodeURIComponent(
            "Hi Nuhad! I am interested in " + product.name
          )
        }"
        target="_blank">
        Ask about this product →
      </a>
    </article>
  `).join("");
}

function openAdmin() {
  if (prompt("Admin password") !== ADMIN_PASSWORD) {
    alert("Incorrect password");
    return;
  }

  modal.classList.add("show");
  drawEditors();
}

function closeAdmin() {
  modal.classList.remove("show");
}

function drawEditors() {
  editors.innerHTML = items.map((product, index) => `
    <div class="editor">
      <div>
        <input
          data-index="${index}"
          data-key="name"
          value="${escapeText(product.name)}"
          placeholder="Product name">

        <input
          data-index="${index}"
          data-key="description"
          value="${escapeText(product.description || "")}"
          placeholder="Short description">
      </div>

      <input
        data-index="${index}"
        data-key="price"
        value="${escapeText(product.price || "")}"
        placeholder="Price">

      <button onclick="removeProduct(${index})">×</button>
    </div>
  `).join("");
}

function addProduct() {
  items.push({
    name: "New product",
    price: "",
    description: ""
  });

  drawEditors();
}

function removeProduct(index) {
  items.splice(index, 1);
  drawEditors();
}

function saveProducts() {
  document.querySelectorAll("#editors input").forEach(input => {
    items[input.dataset.index][input.dataset.key] =
      input.value.trim();
  });

  localStorage.setItem("glow-products", JSON.stringify(items));
  renderProducts();
  closeAdmin();
  alert("Products saved on this device.");
}

year.textContent = new Date().getFullYear();
renderProducts();
</script>
